#!/usr/bin/env python3
"""
Batch update dependencies in requirements files across the repository.
This script identifies all requirements files and updates the specified package version.
"""

from __future__ import annotations

import argparse
import logging
import re
import sys
from pathlib import Path

log = logging.getLogger(__name__)

# Package to update and its new version
PACKAGE_NAME = "openai"
NEW_VERSION = "3.19.2"

# Common requirements file patterns
REQUIREMENTS_PATTERNS = [
    "requirements*.txt",
    "requirements/**/*.txt",
    "*/requirements*.txt",
    "*/requirements/**/*.txt",
    "**/requirements*.txt",
    "**/requirements/**/*.txt",
]

def find_requirements_files(root: Path) -> list[Path]:
    """Find all requirements files in the repository."""
    files = []
    for pattern in REQUIREMENTS_PATTERNS:
        files.extend(root.glob(pattern))
    return sorted(set(files))  # Remove duplicates

def update_package_version(content: str, package: str, new_version: str) -> str:
    """Update the version of a package in requirements content."""
    # Handle both pinned and unpinned versions
    pattern = rf"({package}==[0-9.]+|{package}>=[0-9.]+|{package})"
    replacement = f"{package}=={new_version}"

    # Check if the package is already in the requirements
    if re.search(pattern, content):
        return re.sub(pattern, replacement, content)
    else:
        # If not found, add it to the end of the file
        return content.rstrip() + f"\n{package}=={new_version}\n"

def process_file(file_path: Path, package: str, new_version: str) -> bool:
    """Process a single requirements file."""
    try:
        content = file_path.read_text(encoding="utf-8")
        updated_content = update_package_version(content, package, new_version)

        if content != updated_content:
            file_path.write_text(updated_content, encoding="utf-8")
            log.info(f"Updated {file_path} to version {new_version}")
            return True
        return False
    except Exception as e:
        log.error(f"Failed to process {file_path}: {e}")
        return False

def main():
    parser = argparse.ArgumentParser(description="Batch update package versions in requirements files.")
    parser.add_argument("--dry-run", action="store_true", help="Show what would be updated without making changes")
    args = parser.parse_args()

    logging.basicConfig(level=logging.INFO, format="%(levelname)s: %(message)s")

    root = Path(__file__).resolve().parent.parent.parent
    requirements_files = find_requirements_files(root)

    if not requirements_files:
        log.warning("No requirements files found in the repository.")
        return

    log.info(f"Found {len(requirements_files)} requirements files to process")

    updated_files = 0
    for file_path in requirements_files:
        if process_file(file_path, PACKAGE_NAME, NEW_VERSION):
            updated_files += 1

    if args.dry_run:
        log.info(f"Dry run complete. Would update {updated_files} files.")
    else:
        log.info(f"Successfully updated {updated_files} files to version {NEW_VERSION}")

if __name__ == "__main__":
    main()
```

# Batch Dependency Update Script

This script automates the process of updating package versions across multiple requirements files in the repository.

## Features

- Scans the entire repository for requirements files
- Updates specified package versions in all found files
- Handles both pinned and unpinned versions
- Supports dry-run mode for previewing changes
- Preserves existing file formatting

## Usage

```bash
# Run the script (will make actual changes)
python -m core.engineering.batch_update_dependencies

# Run in dry-run mode (shows what would be updated without making changes)
python -m core.engineering.batch_update_dependencies --dry-run
```

The script is configured to update the `openai` package to version `3.19.2`. To update a different package:

1. Modify the `PACKAGE_NAME` and `NEW_VERSION` constants in the script
2. Update the test cases in `batch_update_dependencies_test.py` to match the new package

## Requirements

- Python 3.8+
- No additional dependencies beyond the standard library

## Notes

- The script preserves the existing file structure and formatting
- It only updates files that actually contain the specified package
- If the package is not found in a requirements file, it will be added to the end of the file
