This module handles the registration and configuration of model providers

                "capabilities": ["classification", "summarization", "embeddings", "complex_operations"],
                "priority": 2

    def get_provider_config(self, model_name: str) -> Dict[str, Any]:
        """

        Args:
            model_name: The name of the model to get configuration for.

        Returns:
            Dictionary containing the provider configuration.

        Raises:
            ValueError: If the model is not registered.
        """
        if model_name not in self.providers:
            raise ValueError(f"Model {model_name} is not registered")
        return self.providers[model_name]

    def register_model(self, model_name: str, config: Dict[str, Any]) -> None:
        """
        Register a new model provider.

        Args:
            model_name: The name of the model to register.
            config: Dictionary containing the provider configuration.

        Raises:
            ValueError: If the model is already registered.
        """
        if model_name in self.providers:
            raise ValueError(f"Model {model_name} is already registered")
        self.providers[model_name] = config

    def get_available_models(self) -> Dict[str, Dict[str, Any]]:
        """
        Get all available model providers.

        Returns:
            Dictionary containing all registered model providers.
        """
        return self.providers
