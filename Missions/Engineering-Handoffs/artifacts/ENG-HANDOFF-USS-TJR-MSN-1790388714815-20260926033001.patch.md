This script processes handoff files in the specified directory, analyzes

def _suggest_status(fields: Dict[str, str]) -> str:
    """Suggest an appropriate status for a handoff based on its content."""

            return "Completed"

        return "In Progress"

def review_handoffs(directory: Path, dry_run: bool = False) -> None:
    """Review handoff files in the specified directory and suggest status updates."""
    if not directory.exists():
        log.error(f"Directory does not exist: {directory}")
        return

    handoff_files = sorted(directory.glob("ENG-HANDOFF-*.md"))
    if not handoff_files:
        log.info(f"No handoff files found in {directory}")
        return

    for path in handoff_files:

        if not fields:
            log.warning(f"Could not parse handoff file: {path}")

                _write_handoff_file(path, fields)

        else:
            log.info(f"No status change needed for {path.name}")
