                "catalogue_entry": {

                    "description": "GLM 5.3 cloud model for document processing",
                    "capabilities": ["classification", "summarization", "embeddings"],
                    "sensitivity_levels": ["standard", "sensitive", "restricted"],
                    "memory_recommendations": ["recommend_full", "recommend_summary_only", "recommend_metadata_only"]
                }
            }
        }

    def get_provider_config(self, model_name: str) -> Optional[Dict]:
        """Get configuration for a specific model provider."""
        return self._provider_configs.get(model_name)

    def validate_credentials(self, model_name: str) -> bool:
        """Validate that credentials are available for the specified model."""
        config = self.get_provider_config(model_name)
        if not config:
            return False

        api_key = os.getenv(config["api_key_env"])
        return api_key is not None and api_key.strip() != ""

    def get_catalogue_entry(self, model_name: str) -> Optional[Dict]:
        """Get the catalogue entry for a specific model."""
        config = self.get_provider_config(model_name)
        return config.get("catalogue_entry") if config else None
