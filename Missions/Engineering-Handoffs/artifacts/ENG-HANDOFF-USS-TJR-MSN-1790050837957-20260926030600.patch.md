    if match:

        parts = content.split("```")
        if len(parts) >= 2:

            inner = inner.strip()

        req = urllib.request.Request(
            f"{self.base_url}{path}", data=payload, method="POST",
            headers={"Content-Type": "application/json"},
        )
        try:
            with urllib.request.urlopen(req, timeout=self.timeout) as resp:  # nosec <US_DRIVER_LICENSE> - base_url is a ctor param, but callers (<URL>) always pass <URL>del_router.url sourced from MODEL_ROUTER_URL env config, not user input - reviewed <DATE_TIME>

            raise ModelRouterError(f"POST {path} failed: {exc.code} {body}") from exc
        except urllib.error.URLError as exc:
            raise ModelRouterError(f"POST {path} unreachable: {exc}") from exc

    def classify_document(self, text: str) -> dict:

            categories=", ".join(CATEGORIES),
            sensitivity=", ".join(SENSITIVITY_LEVELS),
            memory_recs=", ".join(MEMORY_RECOMMENDATIONS),
            text=text[:_CLASSIFY_CHAR_LIMIT],
        )
        result = self._post("/api/model/classify-document", prompt)
        if not result.get("success"):
            raise ModelRouterError(f"classify-document failed: {result.get('error')}")
        parsed = _parse_json_response(result["response"])

        category = parsed.get("category")
        if category not in CATEGORIES:

            memory_recommendation = "needs_review"

            memory_recommendation = "needs_review"  # override for safety

        return {
            "category": category,
            "tags": parsed.get("tags", []),
            "sensitivity": sensitivity,
            "memory_recommendation": memory_recommendation,
            "confidence": parsed.get("confidence", 0.0),

            "summary": parsed.get("summary", ""),

        """Call the model router with the given prompt.

            f"{self.base_url}/api/model/call",

            headers={"Content-Type": "application/json"},

class ModelRouterLatencyAnalyzer:
    """Samples Model Router latency, bucketed by the model label each call
    actually reports (see module docstring — that's the only per-model
    signal available; this cannot force a specific model to be used)."""

    def __init__(self, router_client: ModelRouterClient):
        self.router_client = router_client

    def analyze_latency(
        self,
        prompts: list[str] | None = None,
        samples_per_prompt: int = 1,
    ) -> dict[str, dict[str, float]]:
        """Call the Model Router for each prompt `samples_per_prompt` times,
        recording elapsed time against whichever model label the response
        reports. A failed call is logged and skipped — one bad call must not
        void the rest of the sample."""
        prompts = prompts if prompts is not None else DEFAULT_TEST_PROMPTS
        timings_by_model: dict[str, list[float]] = defaultdict(list)

        for prompt in prompts:
            for _ in range(samples_per_prompt):
                start = time.time()
                try:
                    _, model_label = self.router_client.call(prompt)
                except Exception as exc:  # noqa: BLE001 - best-effort latency sample, already logged
                    log.warning(f"Model router call failed: {exc}")
                    continue
                elapsed = time.time() - start
                timings_by_model[model_label].append(elapsed)
                log.debug(f"model={model_label} elapsed={elapsed:.2f}s")

            model_label: {
