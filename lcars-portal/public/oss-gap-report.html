import asyncio, edge_tts

XO_VOICE = "en-AU-WilliamNeural"

async def speak(text: str, output_path: str) -> str:
    communicate = edge_tts.Communicate(text, XO_VOICE)
    await communicate.save(output_path)
    return output_path
